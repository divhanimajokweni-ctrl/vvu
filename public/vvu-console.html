import type { NextApiRequest, NextApiResponse } from 'next';
import crypto from 'crypto';

const B32 = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';

function b32decode(s: string): Buffer {
  s = s.replace(/=+$/, '').toUpperCase();
  let bits = '';
  for (const c of s) {
    const v = B32.indexOf(c);
    if (v < 0) throw new Error('invalid base32 char: ' + c);
    bits += v.toString(2).padStart(5, '0');
  }
  const n = Buffer.alloc(Math.floor(bits.length / 8));
  for (let i = 0; i < n.length; i++) {
    n.writeUInt8(parseInt(bits.substr(i * 8, 8), 2), i);
  }
  return n;
}

function hotp(secretB32: string, counter: number, digits = 6): string {
  const key = b32decode(secretB32);
  const buf = Buffer.alloc(8);
  buf.writeBigUInt64BE(BigInt(counter));
  const h = crypto.createHmac('sha1', key).update(buf).digest();
  const offset = h[h.length - 1] & 0x0f;
  const code =
    (((h[offset] & 0x7f) << 24) |
      (h[offset + 1] << 16) |
      (h[offset + 2] << 8) |
      h[offset + 3]) % Math.pow(10, digits);
  return String(code).padStart(digits, '0');
}

function totp(secretB32: string, tSec = Math.floor(Date.now() / 1000), step = 30): string {
  return hotp(secretB32, Math.floor(tSec / step));
}

const sessions = new Map<string, { secret: string; userId: string; expires: number }>();

export default function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') { res.status(405).json({ error: 'POST only' }); return; }
  const { action } = req.body || {};

  if (action === 'create') {
    const userId = (req.body?.userId as string) || `user_${crypto.randomUUID().slice(0, 8)}`;
    const secret = crypto.randomBytes(20).toString('hex').toUpperCase()
      .replace(/[^A-Z2-7]/g, '').padEnd(32, 'A').slice(0, 32);
    const otpauth = `otpauth://totp/VVU:${userId}?secret=${secret}&issuer=VVU&period=30&digits=6&algorithm=SHA1`;
    const sessionId = crypto.randomUUID();
    sessions.set(sessionId, { secret, userId, expires: Date.now() + 3600000 });
    res.status(200).json({ sessionId, otpauth, secret, message: 'Add this to Google Authenticator' });
    return;
  }

  if (action === 'verify') {
    const { sessionId, code } = req.body;
    const sess = sessions.get(sessionId);
    if (!sess || sess.expires < Date.now()) {
      sessions.delete(sessionId);
      res.status(401).json({ error: 'session expired' });
      return;
    }
    const tSec = Math.floor(Date.now() / 1000);
    const step = 30;
    for (let w = -1; w <= 1; w++) {
      const expected = totp(sess.secret, (tSec / step + w) * step);
      if (code === expected) {
        const token = crypto.createHmac('sha256', process.env.JWT_SECRET || 'change-me')
          .update(`${sess.userId}${Date.now()}`).digest('hex');
        res.status(200).json({ verified: true, token, userId: sess.userId });
        return;
      }
    }
    res.status(401).json({ error: 'invalid code' });
    return;
  }

  res.status(400).json({ error: 'unknown action' });
}

How ProofBridge secures every transaction.

Three-layer cryptographic attestation. Every contribution, every payout — immutable, auditable.

Under VVU's methodology: (1) Stitch Webhook-secured transactions are receipted in-store with ED25519 keys; (2) CircuitBreaker.sol enforces 3-of-5 oracle quorum on-chain; (3) ANT Telemetry runs Bayesian anomaly detection across log evidence; (4) Reconciliation anchors SHA-256 hashes to Polygon Amoy; (5) FROST threshold signatures provide TEE-backed key custody; (6) All proof objects undergo StableStringify canonicalization before storage.
